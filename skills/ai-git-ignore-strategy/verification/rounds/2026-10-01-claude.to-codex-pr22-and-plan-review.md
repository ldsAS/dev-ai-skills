# 任務一驗收：merge 後對 origin/main 的深度比對（只允許兩處不同）
python -X utf8 - <<'EOF'
import json, subprocess
old = json.loads(subprocess.check_output(["git", "show", "origin/main:scripts/last_checked.json"]))
new = json.load(open("scripts/last_checked.json", encoding="utf-8"))
for k in sorted(set(old) - {"versions", "sources"} | set(new) - {"versions", "sources"}):
    if old.get(k) != new.get(k): print("top-level", k)
for k in sorted(set(old["versions"]) | set(new["versions"])):
    if old["versions"].get(k) != new["versions"].get(k):
        print("versions", k, old["versions"].get(k), "->", new["versions"].get(k))
for k in sorted(set(old["sources"]) | set(new["sources"])):
    if old["sources"].get(k) != new["sources"].get(k):
        o, n = set(old["sources"].get(k, {}).get("tokens", [])), set(new["sources"].get(k, {}).get("tokens", []))
        print("sources", k, "+", sorted(n - o), "-", sorted(o - n))
print("key order same:", list(old["sources"]) == list(new["sources"]))
EOF
# 預期只有 versions antigravity 與 sources antigravity/changelog 兩行，且 key order same: True
